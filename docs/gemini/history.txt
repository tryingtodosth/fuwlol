async def fetch_snapshots_manifest(
    session: aiohttp.ClientSession,
    target_pattern: str,
    from_year: str = "1995",
    to_year: str = "2010"
) -> AsyncGenerator[Dict[str, str], None]:
    params = {
        "url": target_pattern,
        "matchType": "domain",
        "output": "json",
        "fl": "timestamp,original,statuscode,digest,length",
        "filter": ["statuscode:200", "mimetype:text/html"],
        "collapse": "digest",
        "from": f"{from_year}0101",
        "to": f"{to_year}1231",
        "limit": "5000",
        "showResumeKey": "true"
    }
    resume_key = None

    while True:
        req_params = dict(params)
        if resume_key:
            req_params["resumeKey"] = resume_key

        try:
            async with session.get(CDX_URL, params=req_params, timeout=aiohttp.ClientTimeout(total=60)) as resp:
                if resp.status in (429, 503):
                    logger.warning("Limit przepustowości CDX (HTTP %s). Pauza 10s...", resp.status)
                    await asyncio.sleep(10)
                    continue
                resp.raise_for_status()
                payload = await resp.json(content_type=None)
                if not payload:
                    break

                headers = payload[0]
                rows = payload[1:]

                if len(rows) >= 2 and len(rows[-2]) == 0 and len(rows[-1]) == 1:
                    resume_key = rows[-1][0]
                    records = rows[:-2]
                else:
                    resume_key = None
                    records = rows

                for row in records:
                    yield dict(zip(headers, row))

                if not resume_key:
                    break
        except Exception as exc:
            logger.error("Błąd podczas enumeracji CDX: %s", exc)
            break

def decode_rfc2047(header_str: Optional[str]) -> str:
    if not header_str:
        return ""
    fragments = []
    for chunk, encoding in decode_header(header_str):
        if isinstance(chunk, bytes):
            enc = encoding.lower() if encoding else "iso-8859-2"
            try:
                fragments.append(chunk.decode(enc, errors="replace"))
            except (LookupError, UnicodeDecodeError):
                fragments.append(chunk.decode("latin1", errors="replace"))
        else:
            fragments.append(chunk)
    return "".join(fragments)

        return BeautifulSoup("", "html5lib")

    meta_match = META_REGEX.search(raw_bytes[:2048])
    if meta_match:
        try:
            detected_encoding = meta_match.group(1).decode("ascii").lower()
        except UnicodeDecodeError:
            pass

    decoded_string: Optional[str] = None
    priority_list = [detected_encoding] if detected_encoding else []
    priority_list.extend(["utf-8", "iso-8859-2", "windows-1250", "cp852"])

    for enc in priority_list:
        if not enc:
            continue
        try:
            candidate = raw_bytes.decode(enc)
            if any(char in POLISH_PROBE for char in candidate):
                decoded_string = candidate
                break
        except (LookupError, UnicodeDecodeError):
            continue

    if decoded_string is None:
        best_match = charset_normalizer.from_bytes(raw_bytes).best()
        if best_match:
            decoded_string = str(best_match)
        else:
            decoded_string = raw_bytes.decode("iso-8859-2", errors="replace")

    repaired_unicode = ftfy.fix_text(
        decoded_string,
        normalization="NFC",
        fix_character_encodings=True
    )

    return BeautifulSoup(repaired_unicode, "html5lib")

                    "capture_ts": snapshot_ts,
                    "engine": "dom_dl"
                }
                results.append(rec)

    # Ścieżka 2: Analiza struktur tabelarycznych (<table>)
    for table in soup.find_all("table"):
        for tr in table.find_all("tr"):
            cells = [c.get_text(" ", strip=True) for c in tr.find_all(["td", "th"])]
            if len(cells) >= 2:
                joined = " ".join(cells)
                match = SPEAKER_MATCHER.search(cells[0])
                if match:
                    title, name = match.groups()
                    text = cells[-1]
                    context = cells[1] if len(cells) >= 3 else ""
                    if len(text) > 10:
                        rec = {
                            "quote_hash": make_hash(cells[0], text),
                            "academic_title": (title or "").strip(),
                            "speaker": name.strip(),
                            "context": context,
                            "date_raw": "",
                            "quote_text": text,
                            "source_uri": source_url,
                            "capture_ts": snapshot_ts,
                            "engine": "dom_table"
                        }
                        results.append(rec)

    # Ścieżka 3: Analiza bloków tekstowych i list nieuporządkowanych
    for element in soup.find_all(["p", "li", "div"]):
        # Pomijanie węzłów, które zawierają w sobie inne kontenery blokowe
        if element.find(["p", "table", "ul", "dl"]):
            continue
        text_line = element.get_text(" ", strip=True)
        for pattern in PATTERNS:
            match = pattern.search(text_line)
            if match:
                data = match.groupdict()
                speaker_str = data.get("speaker", "").strip()
                sp_match = SPEAKER_MATCHER.search(speaker_str)
                title, name = sp_match.groups() if sp_match else ("", speaker_str)
                quote_text = data.get("quote", "").strip()

                rec = {
                    "quote_hash": make_hash(speaker_str, quote_text),
                    "academic_title": (title or "").strip(),
                    "speaker": (name or speaker_str).strip(),
                    "context": (data.get("context") or "").strip(),
                    "date_raw": (data.get("date") or "").strip(),
                    "quote_text": quote_text,
                    "source_uri": source_url,
                    "capture_ts": snapshot_ts,
                    "engine": "regex_text"
                }
                results.append(rec)
                break

    return results
