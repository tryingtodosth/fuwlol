Moduł normalizacji i kanonizacji w warstwie aplikacjiModuł normalizatora odpowiada za podział treści wejściowej na oczyszczony tekst polski oraz kanoniczny zbiór formuł $\LaTeX$, zapewniając powtarzalność transformacji zarówno podczas zapisu modelu, jak i w momencie przetwarzania parametrów zapytania w widoku API:Pythonimport re

class LatexNormalizer:
    MACRO_SYNONYMS = [
        (re.compile(r'\\dfrac\b'), r'\\frac'),
        (re.compile(r'\\tfrac\b'), r'\\frac'),
        (re.compile(r'\\le\b'), r'\\leq'),
        (re.compile(r'\\ge\b'), r'\\geq'),
        (re.compile(r'\\ne\b'), r'\\neq'),
        (re.compile(r'\\cdot'), r'*'),
    ]
    
    SPACING_MACROS = re.compile(r'\\[,;!]|\\quad\b|\\qquad\b|\\displaystyle\b')
    BRACE_TARGETS = re.compile(r'([\^_])([0-9a-zA-Z])')
    MATH_BLOCKS = re.compile(r'(\$\$.*?\$\$|\$.*?\$|\\\(.*?\\\)|\\\[.*?\\\])', re.DOTALL)

    @classmethod
    def canonicalize_formula(cls, formula: str) -> str:
        clean = cls.SPACING_MACROS.sub('', formula)
        for pattern, replacement in cls.MACRO_SYNONYMS:
            clean = pattern.sub(replacement, clean)
        clean = cls.BRACE_TARGETS.sub(r'\1{\2}', clean)
        # Usunięcie zbędnych spacji z zachowaniem separatorów po komendach bez klamer
        clean = re.sub(r'\s+', '', clean)
        return clean

    @classmethod
    def transform_payload(cls, text: str) -> tuple[str, str]:
        math_segments = []

        def match_handler(match):
            raw_math = match.group(0)
            canonical = cls.canonicalize_formula(raw_math)
            math_segments.append(canonical)
            return ' '

        text_only = cls.MATH_BLOCKS.sub(match_handler, text)
        text_only = re.sub(r'\s+', ' ', text_only).strip()
        math_only = ' '.join(math_segments).strip()
        return text_only, math_only

class EducationalMaterial(models.Model):
    title = models.CharField(max_length=255)
    raw_content = models.TextField()
    
    # Rozdzielone strumienie danych
    prose_content = models.TextField(editable=False)
    canonical_math = models.TextField(editable=False)

    )

            GinIndex(

                name='mat_math_trgm_gin',
                opclasses=['gin_trgm_ops']
            ),
        ]

from .models import EducationalMaterial

class MaterialSerializer(serializers.ModelSerializer):
    class Meta:
        model = EducationalMaterial
        fields = ['id', 'title', 'raw_content']

class UnifiedSearchAPIView(APIView):
    def get(self, request):
        query_text = request.query_params.get('q', '').strip()
        if not query_text:
            return Response([], status=status.HTTP_200_OK)

        text_query, math_query = LatexNormalizer.transform_payload(query_text)
        queryset = EducationalMaterial.objects.all()

        has_text = bool(text_query)
        has_math = bool(math_query)

        # Ewaluacja lingwistyczna (język polski)
        if has_text:
            polish_query = SearchQuery(text_query, config='polish', search_type='websearch')
            queryset = queryset.filter(search_vector=polish_query).annotate(
                text_score=SearchRank(F('search_vector'), polish_query)
            )
        else:
            queryset = queryset.annotate(text_score=Value(0.0, output_field=FloatField()))

        # Ewaluacja matematyczna (podwyrażenia LaTeX)
        if has_math:
            queryset = queryset.filter(canonical_math__trigram_similar=math_query).annotate(
                math_score=TrigramSimilarity('canonical_math', math_query)
            )
        else:
            queryset = queryset.annotate(math_score=Value(0.0, output_field=FloatField()))

        if not has_text and not has_math:
            return Response([], status=status.HTTP_200_OK)

        # Fuzja rankingów: wyższy priorytet dla precyzyjnego trafienia wzoru
        queryset = queryset.annotate(
            final_rank=F('text_score') + (F('math_score') * 2.0)
        ).order_by('-final_rank')[:30]

        serializer = MaterialSerializer(queryset, many=True)
        return Response(serializer.data, status=status.HTTP_200_OK)

Zastosowanie wbudowanego modułu SQLite FTS5 z zewnętrznym stemmerem w Pythonie (pystempel) zapewnia minimalny narzut pamięciowy, lecz wymusza rezygnację z natywnego ORM Django oraz ręczne konstruowanie struktur bazodanowych w surowym SQL.W konsekwencji najkorzystniejszym rozwiązaniem pod względem stabilności, poprawności lingwistycznej oraz czystości architektury jest migracja bazy danych na PostgreSQL. Baza ta, skonfigurowana ze słownikiem Hunspell (pl_pl) oraz rozszerzeniem pg_trgm, z powodzeniem mieści się w budżecie pamięciowym niewielkiego serwera VPS, rozwiązując problem wyszukiwania w fleksyjnym języku polskim oraz umożliwiając precyzyjne odnajdywanie zkanonizowanych podwzorów matematycznych w ramach jednego, zunifikowanego procesu bazodanowego.
